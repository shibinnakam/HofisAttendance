ESP8266 / ESP32 + RC522 RFID Integration Code

Flash this sketch to your NodeMCU / ESP8266 / ESP32 connected to an MFRC522 RFID Reader. It reads RFID cards and automatically posts the UID to your backend endpoint: /api/attendance/tap. 

// RFID Attendance System - ESP8266 / ESP32 + MFRC522
#include <SPI.h>
#include <MFRC522.h>
#if defined(ESP8266)
  #include <ESP8266WiFi.h>
  #include <ESP8266HTTPClient.h>
#else
  #include <WiFi.h>
  #include <HTTPClient.h>
#endif

// Wi-Fi Configuration
const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";

// Node.js Backend URL (Replace with your local IP or deployed server URL)
const char* serverUrl = "http://192.168.1.100:5000/api/attendance/tap";

#define SS_PIN  D8  // or GPIO 5 on ESP32
#define RST_PIN D3  // or GPIO 22 on ESP32
MFRC522 rfid(SS_PIN, RST_PIN);

void setup() {
  Serial.begin(115200);
  SPI.begin();
  rfid.PCD_Init();
  
  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nConnected! Ready to scan RFID cards...");
}

void loop() {
  if (!rfid.PICC_IsNewCardPresent() || !rfid.PICC_ReadCardSerial()) return;

  // Convert UID bytes to Hex String
  String cardUID = "";
  for (byte i = 0; i < rfid.uid.size; i++) {
    if (rfid.uid.uidByte[i] < 0x10) cardUID += "0";
    cardUID += String(rfid.uid.uidByte[i], HEX);
  }
  cardUID.toUpperCase();
  Serial.println("\n[RFID Tapped] UID: " + cardUID);

  // Send POST request to backend
  if (WiFi.status() == WL_CONNECTED) {
    WiFiClient client;
    HTTPClient http;
    http.begin(client, serverUrl);
    http.addHeader("Content-Type", "application/json");

    String payload = "{\"rfidCardNumber\":\"" + cardUID + "\"}";
    int httpCode = http.POST(payload);
    String response = http.getString();

    Serial.print("HTTP Response: ");
    Serial.println(httpCode);
    Serial.println(response);
    http.end();
  }

  rfid.PICC_HaltA();
  rfid.PCD_StopCrypto1();
  delay(1500); // Prevent duplicate rapid reads
}
